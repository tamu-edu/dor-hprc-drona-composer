#!/bin/bash
#SBATCH --job-name=[JOBNAME]
#SBATCH --time=00:10:00 --mem=1G
#SBATCH --ntasks=1 --cpus-per-task=1
#SBATCH --output=out.%j --error=error.%j

cd [flocation]

tr -cs '[:alpha:]' '\n' < "[INPUT_FILE]" \
  | tr '[:upper:]' '[:lower:]' \
  | grep . \
  [FILTER] \
  | sort | uniq -c | sort -rn \
  | head -n [TOP_N] > results.txt
